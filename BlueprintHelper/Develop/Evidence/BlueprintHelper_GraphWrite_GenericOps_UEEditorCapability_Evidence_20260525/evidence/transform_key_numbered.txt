  2169	const FPinConnectionResponse UEdGraphSchema_K2::CanCreateConnection(const UEdGraphPin* PinA, const UEdGraphPin* PinB) const
  2170	{
  2171		check(PinA);
  2172		check(PinB);
  2173	
  2174		const UK2Node* OwningNodeA = Cast<UK2Node>(PinA->GetOwningNodeUnchecked());
  2175		const UK2Node* OwningNodeB = Cast<UK2Node>(PinB->GetOwningNodeUnchecked());
  2176	
  2177		if (!OwningNodeA || !OwningNodeB)
  2178		{
  2179			return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, TEXT("Invalid nodes"));
  2180		}
  2181	
  2182		// Make sure the pins are not on the same node
  2183		if (OwningNodeA == OwningNodeB)
  2184		{
  2185			return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, TEXT("Both are on the same node"));
  2186		}
  2187	
  2188		if (PinA->bOrphanedPin || PinB->bOrphanedPin)
  2189		{
  2190			return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, TEXT("Cannot make new connections to orphaned pin"));
  2191		}
  2192	
  2193		FString NodeResponseMessage;
  2194		// node can disallow the connection
  2195		{
  2196			if(OwningNodeA && OwningNodeA->IsConnectionDisallowed(PinA, PinB, NodeResponseMessage))
  2197			{
  2198				return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, NodeResponseMessage);
  2199			}
  2200			if(OwningNodeB && OwningNodeB->IsConnectionDisallowed(PinB, PinA, NodeResponseMessage))
  2201			{
  2202				return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, NodeResponseMessage);
  2203			}
  2204		}
  2205	
  2206		// Compare the directions
  2207		const UEdGraphPin* InputPin = nullptr;
  2208		const UEdGraphPin* OutputPin = nullptr;
  2209	
  2210		if (!CategorizePinsByDirection(PinA, PinB, /*out*/ InputPin, /*out*/ OutputPin))
  2211		{
  2212			return FPinConnectionResponse(CONNECT_RESPONSE_DISALLOW, TEXT("Directions are not compatible"));
  2213		}
  2214	
  2215		check(InputPin);
  2216		check(OutputPin);
  2217	
  2218		bool bIgnoreArray = false;
  2219		if (const UK2Node* OwningNode = Cast<UK2Node>(InputPin->GetOwningNode()))
  2220		{
  2221			const bool bAllowMultipleSelfs = OwningNode->AllowMultipleSelfs(true); // it applies also to ForEachCall
  2222			const bool bNotAContainer = !InputPin->PinType.IsContainer();
  2223			const bool bSelfPin = IsSelfPin(*InputPin);
  2224			if (bAllowMultipleSelfs && bNotAContainer && bSelfPin)
  2225			{
  2226				// Indicates whether or not we will allow an array to be connected to a non-array input. This applies to nodes that support a foreach expansion of array inputs.
  2227				bIgnoreArray = OutputPin->PinType.IsArray();
  2228			}
  2229		}
  2230	
  2231		// Find the calling context in case one of the pins is of type object and has a value of Self
  2232		UClass* CallingContext = nullptr;
  2233		const UBlueprint* Blueprint = FBlueprintEditorUtils::FindBlueprintForNode(PinA->GetOwningNodeUnchecked());
  2234		if (Blueprint)
  2235		{
  2236			CallingContext = (Blueprint->GeneratedClass != NULL) ? Blueprint->GeneratedClass : Blueprint->ParentClass;
  2237		}
  2238	
  2239		// Compare the types
  2240		const bool bTypesMatch = ArePinsCompatible(OutputPin, InputPin, CallingContext, bIgnoreArray);
  2241	
  2242		if (bTypesMatch)
  2243		{
  2244			FPinConnectionResponse ConnectionResponse = DetermineConnectionResponseOfCompatibleTypedPins(PinA, PinB, InputPin, OutputPin);
  2245			if (ConnectionResponse.Message.IsEmpty())
  2246			{
  2247				ConnectionResponse.Message = FText::FromString(NodeResponseMessage);
  2248			}
  2249			else if (!NodeResponseMessage.IsEmpty())
  2250			{
  2251				ConnectionResponse.Message = FText::Format(LOCTEXT("MultiMsgConnectionResponse", "{0} - {1}"), ConnectionResponse.Message, FText::FromString(NodeResponseMessage));
  2252			}
  2253			return ConnectionResponse;
  2254		}
  2255		else
  2256		{
  2257			// Promotable types in blueprints! Only if the Cvar is set and the node is of a special type. Eventually we want this for all
  2258			if (TypePromoDebug::IsTypePromoEnabled() && InputPin->GetOwningNode()->IsA<UK2Node_PromotableOperator>())
  2259			{
  2260				if (FTypePromotion::IsValidPromotion(InputPin->PinType, OutputPin->PinType) || FTypePromotion::HasStructConversion(InputPin, OutputPin))
  2261				{
  2262					// Set the Text here correctly based on which pin type is higher
  2263					return FPinConnectionResponse(CONNECT_RESPONSE_MAKE_WITH_PROMOTION, FString::Printf(TEXT("Promote %s to %s"), *TypeToText(InputPin->PinType).ToString(), *TypeToText(OutputPin->PinType).ToString()));
  2264				}
  2265			}
  2266	
  2267			// Autocasting
  2268			const bool bCanAutocast = SearchForAutocastFunction(OutputPin->PinType, InputPin->PinType).IsSet();
  2269			const bool bCanAutoConvert = FindSpecializedConversionNode(OutputPin->PinType, *InputPin, false).IsSet();
  2270	
  2271			if (bCanAutocast || bCanAutoConvert)
  2272			{
  2273				return FPinConnectionResponse(CONNECT_RESPONSE_MAKE_WITH_CONVERSION_NODE, FString::Printf(TEXT("Convert %s to %s"), *TypeToText(OutputPin->PinType).ToString(), *TypeToText(InputPin->PinType).ToString()));
  2274			}
  2275			else
  2276			{
  2277				bool bIsFatal = true;
  2278				FText IncompatibilityReasonText = GetPinIncompatibilityReason(PinA, PinB, &bIsFatal);
  2279	
  2280				FPinConnectionResponse ConnectionResponse(CONNECT_RESPONSE_DISALLOW, IncompatibilityReasonText.ToString());
  2281				if (bIsFatal)
  2282				{
  2283					ConnectionResponse.SetFatal();
  2284				}
  2285				return ConnectionResponse;
  2286			}
  2287		}

  2741		// If connecting an object to a 'call function' self pin, and not currently compatible, see if there is a property we can call a function on
  2742		else if (bTryAlternateObjectProperty)
  2743		{
  2744			const UK2Node_CallFunction* CallFunctionNode = CastChecked<UK2Node_CallFunction>(InputPin.GetOwningNode());
  2745			const UClass* OutputPinClass = Cast<UClass>(OutputPinType.PinSubCategoryObject.Get());
  2746			const UClass* FunctionClass = CallFunctionNode->FunctionReference.GetMemberParentClass(CallFunctionNode->GetBlueprintClassFromNode());
  2747	
  2748			if(FunctionClass != nullptr && OutputPinClass != nullptr)
  2749			{
  2750				// Iterate over object properties..
  2751				for (TFieldIterator<FObjectProperty> PropIt(OutputPinClass, EFieldIteratorFlags::IncludeSuper); PropIt; ++PropIt)
  2752				{
  2753					FObjectProperty* ObjProp = *PropIt;
  2754					// .. if we have a blueprint visible var, and is of the type which contains this function..
  2755					if(ObjProp->HasAllPropertyFlags(CPF_BlueprintVisible) && ObjProp->PropertyClass && ObjProp->PropertyClass->IsChildOf(FunctionClass))
  2756					{
  2757						Result = { nullptr };
  2758						if (bCreateNode)
  2759						{
  2760							UK2Node_VariableGet* GetNode = NewObject<UK2Node_VariableGet>();
  2761							GetNode->VariableReference.SetFromField<FProperty>(ObjProp, false);
  2762							Result->TargetNode = GetNode;
  2763						}
  2764					}
  2765				}
  2766	
  2767			}	
  2768		}
  2769	
  2770		if (!Result.IsSet())
  2771		{
  2772			// CHECK ENUM TO NAME CAST
  2773			const bool bInputMatch = !InputPin.PinType.IsContainer() && ((PC_Name == InputPin.PinType.PinCategory) || (PC_String == InputPin.PinType.PinCategory));
  2774			const bool bOutputMatch = !OutputPinType.IsContainer() && (PC_Byte == OutputPinType.PinCategory) && (nullptr != Cast<UEnum>(OutputPinType.PinSubCategoryObject.Get()));
  2775			if(bOutputMatch && bInputMatch)
  2776			{
  2777				Result = { nullptr };
  2778				if (bCreateNode)
  2779				{
  2780					if(PC_Name == InputPin.PinType.PinCategory)
  2781					{
  2782						Result->TargetNode = NewObject<UK2Node_GetEnumeratorName>();
  2783					}
  2784					else if(PC_String == InputPin.PinType.PinCategory)
  2785					{
  2786						Result->TargetNode = NewObject<UK2Node_GetEnumeratorNameAsString>();
  2787					}
  2788				}
  2789			}
  2790		}
  2791	
  2792		if (!Result.IsSet())
  2793		{
  2794			// CHECK BYTE TO ENUM CAST
  2795			UEnum* Enum = Cast<UEnum>(InputPinType.PinSubCategoryObject.Get());
  2796			const bool bInputIsEnum = !InputPinType.IsContainer() && (PC_Byte == InputPinType.PinCategory) && Enum;
  2797			const bool bOutputIsByte = !OutputPinType.IsContainer() && (PC_Byte == OutputPinType.PinCategory);
  2798			if (bInputIsEnum && bOutputIsByte)
  2799			{
  2800				Result = { nullptr };
  2801				if(bCreateNode)
  2802				{
  2803					UK2Node_CastByteToEnum* CastByteToEnum = NewObject<UK2Node_CastByteToEnum>();
  2804					CastByteToEnum->Enum = Enum;
  2805					CastByteToEnum->bSafe = true;
  2806					Result->TargetNode = CastByteToEnum;
  2807				}
  2808			}
  2809			else if (!OutputPinType.IsContainer())
  2810			{
  2811				// Note: Cast nodes do not support a ForEach-style expansion, so we exclude container types here.
  2812			
  2813				const UBlueprint* Blueprint = FBlueprintEditorUtils::FindBlueprintForNodeChecked(InputPin.GetOwningNode());
  2814				UClass* BlueprintClass = (Blueprint->GeneratedClass != nullptr) ? Blueprint->GeneratedClass : Blueprint->ParentClass;
  2815	
  2816				UClass* InputClass = Cast<UClass>(InputPin.PinType.PinSubCategoryObject.Get());
  2817	
  2818				if ((InputClass == nullptr) && (InputPin.PinType.PinSubCategory == UEdGraphSchema_K2::PSC_Self))
  2819				{
  2820					InputClass = BlueprintClass;
  2821				}
  2822	
  2823				const UClass* OutputClass = Cast<UClass>(OutputPinType.PinSubCategoryObject.Get());
  2824	
  2825				if ((OutputClass == nullptr) && (OutputPinType.PinSubCategory == UEdGraphSchema_K2::PSC_Self))
  2826				{
  2827					OutputClass = BlueprintClass;
  2828				}
  2829	
  2830				bool bNeedsDynamicCast = false;
  2831				if ((OutputPinType.PinCategory == PC_Interface) && (InputPinType.PinCategory == PC_Object))
  2832				{
  2833					bNeedsDynamicCast = (InputClass && OutputClass) && (InputClass->ImplementsInterface(OutputClass) || OutputClass->IsChildOf(InputClass));
  2834				}
  2835				else if (OutputPinType.PinCategory == PC_Object)
  2836				{
  2837					UBlueprintEditorSettings const* BlueprintSettings = GetDefault<UBlueprintEditorSettings>();
  2838					if ((InputPinType.PinCategory == PC_Object) && BlueprintSettings->bAutoCastObjectConnections)
  2839					{
  2840						bNeedsDynamicCast = (InputClass && OutputClass) && InputClass->IsChildOf(OutputClass);
  2841					}
  2842				}
  2843	
  2844				if (bNeedsDynamicCast)
  2845				{
  2846					Result = { nullptr };
  2847					if (bCreateNode)
  2848					{
  2849						UK2Node_DynamicCast* DynCastNode = NewObject<UK2Node_DynamicCast>();
  2850						DynCastNode->TargetType = InputClass;
  2851						DynCastNode->SetPurity(true);
  2852						Result->TargetNode = DynCastNode;
  2853					}
  2854				}
  2855	
  2856				if (!bNeedsDynamicCast && InputClass && OutputClass && OutputClass->IsChildOf(InputClass))
  2857				{
  2858					const bool bConvertAsset = (OutputPinType.PinCategory == PC_SoftObject) && (InputPinType.PinCategory == PC_Object);
  2859					const bool bConvertAssetClass = (OutputPinType.PinCategory == PC_SoftClass) && (InputPinType.PinCategory == PC_Class);
  2860					const bool bConvertToAsset = (OutputPinType.PinCategory == PC_Object) && (InputPinType.PinCategory == PC_SoftObject);
  2861					const bool bConvertToAssetClass = (OutputPinType.PinCategory == PC_Class) && (InputPinType.PinCategory == PC_SoftClass);
  2862	
  2863					if (bConvertAsset || bConvertAssetClass || bConvertToAsset || bConvertToAssetClass)
  2864					{
  2865						Result = { nullptr };
  2866						if (bCreateNode)
  2867						{
  2868							UK2Node_ConvertAsset* ConvertAssetNode = NewObject<UK2Node_ConvertAsset>();
  2869							Result->TargetNode = ConvertAssetNode;
  2870						}
  2871					}
  2872				}
  2873			}
  2874		}
  2875	
  2876		return Result;
  2877	}

  2879	void UEdGraphSchema_K2::AutowireConversionNode(UEdGraphPin* InputPin, UEdGraphPin* OutputPin, UEdGraphNode* ConversionNode) const
  2880	{
  2881		bool bAllowInputConnections = true;
  2882		bool bAllowOutputConnections = true;
  2883	
  2884		for (int32 PinIndex = 0; PinIndex < ConversionNode->Pins.Num(); ++PinIndex)
  2885		{
  2886			UEdGraphPin* TestPin = ConversionNode->Pins[PinIndex];
  2887	
  2888			UClass* Context = nullptr;
  2889			UK2Node* K2Node = Cast<UK2Node>(OutputPin->GetOwningNode());
  2890			if (K2Node != nullptr)
  2891			{
  2892				UBlueprint* Blueprint = K2Node->GetBlueprint();
  2893				if (Blueprint)
  2894				{
  2895					Context = Blueprint->GeneratedClass;
  2896				}
  2897			}
  2898	
  2899			if ((TestPin->Direction == EGPD_Input) && (ArePinTypesCompatible(OutputPin->PinType, TestPin->PinType, Context)))
  2900			{
  2901				if(bAllowOutputConnections && TryCreateConnection(TestPin, OutputPin))
  2902				{
  2903					// Successful connection, do not allow more output connections
  2904					bAllowOutputConnections = false;
  2905				}
  2906			}
  2907			else if ((TestPin->Direction == EGPD_Output) && (ArePinTypesCompatible(TestPin->PinType, InputPin->PinType, Context)))
  2908			{
  2909				if(bAllowInputConnections && TryCreateConnection(TestPin, InputPin))
  2910				{
  2911					// Successful connection, do not allow more input connections
  2912					bAllowInputConnections = false;
  2913				}
  2914			}
  2915		}

  2918	bool UEdGraphSchema_K2::CreateAutomaticConversionNodeAndConnections(UEdGraphPin* PinA, UEdGraphPin* PinB) const
  2919	{
  2920		// Determine which pin is an input and which pin is an output
  2921		UEdGraphPin* InputPin = nullptr;
  2922		UEdGraphPin* OutputPin = nullptr;
  2923		if (!CategorizePinsByDirection(PinA, PinB, /*out*/ InputPin, /*out*/ OutputPin))
  2924		{
  2925			return false;
  2926		}
  2927	
  2928		check(InputPin);
  2929		check(OutputPin);
  2930	
  2931		UK2Node* TemplateConversionNode = nullptr;
  2932	
  2933		if (TOptional<FSearchForAutocastFunctionResults> AutocastResult = SearchForAutocastFunction(OutputPin->PinType, InputPin->PinType))
  2934		{
  2935			// Create a new call function node for the casting operator
  2936			UK2Node_CallFunction* TemplateNode = NewObject<UK2Node_CallFunction>();
  2937			TemplateNode->FunctionReference.SetExternalMember(AutocastResult->TargetFunction, AutocastResult->FunctionOwner);
  2938	
  2939			TemplateConversionNode = TemplateNode;
  2940		}
  2941		else if (TOptional<FFindSpecializedConversionNodeResults> ConversionResult = FindSpecializedConversionNode(OutputPin->PinType, *InputPin, true))
  2942		{
  2943			TemplateConversionNode = ConversionResult->TargetNode;
  2944		}
  2945	
  2946		if (TemplateConversionNode != nullptr)
  2947		{
  2948			// Determine where to position the new node (assuming it isn't going to get beaded)
  2949			FVector2D AverageLocation = CalculateAveragePositionBetweenNodes(InputPin, OutputPin);
  2950	
  2951			UK2Node* ConversionNode = FEdGraphSchemaAction_K2NewNode::SpawnNodeFromTemplate<UK2Node>(InputPin->GetOwningNode()->GetGraph(), TemplateConversionNode, AverageLocation);
  2952	
  2953			// Connect the cast node up to the output/input pins
  2954			AutowireConversionNode(InputPin, OutputPin, ConversionNode);
  2955	
  2956			return true;
  2957		}
  2958	
  2959		return false;
  2960	}

    80		return SourcePin ? (SourcePin->PinType.PinCategory == UEdGraphSchema_K2::PC_Class || SourcePin->PinType.PinCategory == UEdGraphSchema_K2::PC_Object) : false;
    81	}

    83	bool UK2Node_ConvertAsset::IsConnectionDisallowed(const UEdGraphPin* MyPin, const UEdGraphPin* OtherPin, FString& OutReason) const
    84	{
    85		UEdGraphPin* InputPin = FindPin(UK2Node_ConvertAssetImpl::InputPinName);
    86		if (InputPin && OtherPin && (InputPin == MyPin) && (MyPin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard))
    87		{
    88			if ((OtherPin->PinType.PinCategory != UEdGraphSchema_K2::PC_SoftObject) &&
    89				(OtherPin->PinType.PinCategory != UEdGraphSchema_K2::PC_SoftClass) &&
    90				(OtherPin->PinType.PinCategory != UEdGraphSchema_K2::PC_Object) &&
    91				(OtherPin->PinType.PinCategory != UEdGraphSchema_K2::PC_Class))
    92			{
    93				return true;
    94			}
    95		}
    96		return false;
    97	}

    99	void UK2Node_ConvertAsset::RefreshPinTypes()
   100	{
   101		const UEdGraphSchema_K2* K2Schema = CastChecked<UEdGraphSchema_K2>(GetSchema());
   102		UEdGraphPin* InputPin = FindPin(UK2Node_ConvertAssetImpl::InputPinName);
   103		UEdGraphPin* OutputPin = FindPin(UK2Node_ConvertAssetImpl::OutputPinName);
   104		ensure(InputPin && OutputPin);
   105		if (InputPin && OutputPin)
   106		{
   107			const bool bIsConnected = InputPin->LinkedTo.Num() > 0;
   108			UClass* TargetType = bIsConnected ? GetTargetClass() : nullptr;
   109			const bool bIsClassType = bIsConnected ? IsClassType() : false;
   110			const bool bIsConvertToSoft = bIsConnected ? IsConvertToSoft() : false;
   111	
   112			FName InputCategory = UEdGraphSchema_K2::PC_Wildcard;
   113			FName OutputCategory = UEdGraphSchema_K2::PC_Wildcard;
   114			if (bIsConnected)
   115			{
   116				if (bIsConvertToSoft)
   117				{
   118					InputCategory = (bIsClassType ? UEdGraphSchema_K2::PC_Class : UEdGraphSchema_K2::PC_Object);
   119					OutputCategory = (bIsClassType ? UEdGraphSchema_K2::PC_SoftClass : UEdGraphSchema_K2::PC_SoftObject);
   120				}
   121				else
   122				{
   123					InputCategory = (bIsClassType ? UEdGraphSchema_K2::PC_SoftClass : UEdGraphSchema_K2::PC_SoftObject);
   124					OutputCategory = (bIsClassType ? UEdGraphSchema_K2::PC_Class : UEdGraphSchema_K2::PC_Object);
   125				}
   126			}
   127				
   128			InputPin->PinType = FEdGraphPinType(InputCategory, NAME_None, TargetType, EPinContainerType::None, false, FEdGraphTerminalType() );
   129			OutputPin->PinType = FEdGraphPinType(OutputCategory, NAME_None, TargetType, EPinContainerType::None, false, FEdGraphTerminalType() );
   130	
   131			PinTypeChanged(InputPin);
   132			PinTypeChanged(OutputPin);
   133	
   134			if (OutputPin->LinkedTo.Num())
   135			{
   136				TArray<UEdGraphPin*> PinsToUnlink = OutputPin->LinkedTo;
   137	
   138				UClass const* CallingContext = nullptr;
   139				if (UBlueprint const* Blueprint = GetBlueprint())
   140				{
   141					CallingContext = Blueprint->GeneratedClass;
   142					if (CallingContext == NULL)
   143					{
   144						CallingContext = Blueprint->ParentClass;
   145					}
   146				}
   147	
   148				for (UEdGraphPin* TargetPin : PinsToUnlink)
   149				{
   150					if (TargetPin && !K2Schema->ArePinsCompatible(OutputPin, TargetPin, CallingContext))
   151					{
   152						OutputPin->BreakLinkTo(TargetPin);
   153					}
   154				}
   155			}
   156		}
   157	}
