   439	void UK2Node_SpawnActorFromClass::ExpandNode(class FKismetCompilerContext& CompilerContext, UEdGraph* SourceGraph)
   440	{
   441		Super::ExpandNode(CompilerContext, SourceGraph);
   442	
   443		// make sure that if scale method needs to be given a default value, it gets it
   444		FixupScaleMethodPin();
   445	
   446		static const FName BeginSpawningBlueprintFuncName = GET_FUNCTION_NAME_CHECKED(UGameplayStatics, BeginDeferredActorSpawnFromClass);
   447		static const FName ActorClassParamName(TEXT("ActorClass"));
   448		static const FName WorldContextParamName(TEXT("WorldContextObject"));
   449	
   450		static const FName FinishSpawningFuncName = GET_FUNCTION_NAME_CHECKED(UGameplayStatics, FinishSpawningActor);
   451		static const FName ActorParamName(TEXT("Actor"));
   452		static const FName TransformParamName(TEXT("SpawnTransform"));
   453		static const FName CollisionHandlingOverrideParamName(TEXT("CollisionHandlingOverride"));
   454		static const FName OwnerParamName(TEXT("Owner"));
   455	
   456		static const FName ObjectParamName(TEXT("Object"));
   457		static const FName ValueParamName(TEXT("Value"));
   458		static const FName PropertyNameParamName(TEXT("PropertyName"));
   459	
   460		UK2Node_SpawnActorFromClass* SpawnNode = this;
   461		UEdGraphPin* SpawnNodeExec = SpawnNode->GetExecPin();
   462		UEdGraphPin* SpawnNodeTransform = SpawnNode->GetSpawnTransformPin();
   463		UEdGraphPin* SpawnNodeCollisionHandlingOverride = GetCollisionHandlingOverridePin();
   464		UEdGraphPin* SpawnNodeScaleMethod = TryGetScaleMethodPin();
   465		UEdGraphPin* SpawnWorldContextPin = SpawnNode->GetWorldContextPin();
   466		UEdGraphPin* SpawnClassPin = SpawnNode->GetClassPin();
   467		UEdGraphPin* SpawnNodeOwnerPin = SpawnNode->GetOwnerPin();
   468		UEdGraphPin* SpawnNodeThen = SpawnNode->GetThenPin();
   469		UEdGraphPin* SpawnNodeResult = SpawnNode->GetResultPin();
   470	
   471		// Cache the class to spawn. Note, this is the compile time class that the pin was set to or the variable type it was connected to. Runtime it could be a child.
   472		UClass* ClassToSpawn = GetClassToSpawn();
   473	
   474		UClass* SpawnClass = (SpawnClassPin != NULL) ? Cast<UClass>(SpawnClassPin->DefaultObject) : NULL;
   475		if ( !SpawnClassPin || ((0 == SpawnClassPin->LinkedTo.Num()) && (NULL == SpawnClass)))
   476		{
   477			CompilerContext.MessageLog.Error(*LOCTEXT("SpawnActorNodeMissingClass_Error", "Spawn node @@ must have a @@ specified.").ToString(), SpawnNode, SpawnClassPin);
   478			// we break exec links so this is the only error we get, don't want the SpawnActor node being considered and giving 'unexpected node' type warnings
   479			SpawnNode->BreakAllNodeLinks();
   480			return;
   481		}
   482	
   483		//////////////////////////////////////////////////////////////////////////
   484		// create 'begin spawn' call node
   485		UK2Node_CallFunction* CallBeginSpawnNode = CompilerContext.SpawnIntermediateNode<UK2Node_CallFunction>(SpawnNode, SourceGraph);
   486		CallBeginSpawnNode->FunctionReference.SetExternalMember(BeginSpawningBlueprintFuncName, UGameplayStatics::StaticClass());
   487		CallBeginSpawnNode->AllocateDefaultPins();
   488	
   489		UEdGraphPin* CallBeginExec = CallBeginSpawnNode->GetExecPin();
   490		UEdGraphPin* CallBeginWorldContextPin = CallBeginSpawnNode->FindPinChecked(WorldContextParamName);
   491		UEdGraphPin* CallBeginActorClassPin = CallBeginSpawnNode->FindPinChecked(ActorClassParamName);
   492		UEdGraphPin* CallBeginTransform = CallBeginSpawnNode->FindPinChecked(TransformParamName);
   493		UEdGraphPin* CallBeginScaleMethod = CallBeginSpawnNode->FindPinChecked(FK2Node_SpawnActorFromClassHelper::TransformScaleMethodPinName);
   494		UEdGraphPin* CallBeginCollisionHandlingOverride = CallBeginSpawnNode->FindPinChecked(CollisionHandlingOverrideParamName);
   495	
   496		UEdGraphPin* CallBeginOwnerPin = CallBeginSpawnNode->FindPinChecked(FK2Node_SpawnActorFromClassHelper::OwnerPinName);
   497		UEdGraphPin* CallBeginResult = CallBeginSpawnNode->GetReturnValuePin();	
   498	
   499		// Move 'exec' connection from spawn node to 'begin spawn'
   500		CompilerContext.MovePinLinksToIntermediate(*SpawnNodeExec, *CallBeginExec);
   501	
   502		if(SpawnClassPin->LinkedTo.Num() > 0)
   503		{
   504			// Copy the 'blueprint' connection from the spawn node to 'begin spawn'
   505			CompilerContext.MovePinLinksToIntermediate(*SpawnClassPin, *CallBeginActorClassPin);
   506		}
   507		else
   508		{
   509			// Copy blueprint literal onto begin spawn call 
   510			CallBeginActorClassPin->DefaultObject = SpawnClass;
   511		}
   512	
   513		// Copy the world context connection from the spawn node to 'begin spawn' if necessary
   514		if (SpawnWorldContextPin)
   515		{
   516			CompilerContext.MovePinLinksToIntermediate(*SpawnWorldContextPin, *CallBeginWorldContextPin);
   517		}
   518	
   519		if (SpawnNodeOwnerPin != nullptr)
   520		{
   521			CompilerContext.MovePinLinksToIntermediate(*SpawnNodeOwnerPin, *CallBeginOwnerPin);
   522		}
   523	
   524		// Copy the 'transform' connection from the spawn node to 'begin spawn'
   525		CompilerContext.MovePinLinksToIntermediate(*SpawnNodeTransform, *CallBeginTransform);
   526	
   527		// Copy the 'CollisionHandlingOverride' connection from the spawn node to 'begin spawn'
   528		CompilerContext.MovePinLinksToIntermediate(*SpawnNodeCollisionHandlingOverride, *CallBeginCollisionHandlingOverride);
   529	
   530		// Copy the 'ScaleMethod' connection from the spawn node to 'begin spawn'
   531		if (SpawnNodeScaleMethod)
   532		{
   533			CompilerContext.MovePinLinksToIntermediate(*SpawnNodeScaleMethod, *CallBeginScaleMethod);
   534		}
   535		else
   536		{
   537			// if for some reason we couldn't find scale method pin on the spawn node, defer it's value to runtime
   538		    const UEnum* const ScaleMethodEnum = StaticEnum<ESpawnActorScaleMethod>();
   539			CallBeginScaleMethod->DefaultValue = ScaleMethodEnum->GetNameStringByValue(static_cast<int>(ESpawnActorScaleMethod::SelectDefaultAtRuntime));
   540		}
   541	
   542		//////////////////////////////////////////////////////////////////////////
   543		// create 'finish spawn' call node
   544		UK2Node_CallFunction* CallFinishSpawnNode = CompilerContext.SpawnIntermediateNode<UK2Node_CallFunction>(SpawnNode, SourceGraph);
   545		CallFinishSpawnNode->FunctionReference.SetExternalMember(FinishSpawningFuncName, UGameplayStatics::StaticClass());
   546		CallFinishSpawnNode->AllocateDefaultPins();
   547	
   548		UEdGraphPin* CallFinishExec = CallFinishSpawnNode->GetExecPin();
   549		UEdGraphPin* CallFinishThen = CallFinishSpawnNode->GetThenPin();
   550		UEdGraphPin* CallFinishActor = CallFinishSpawnNode->FindPinChecked(ActorParamName);
   551		UEdGraphPin* CallFinishTransform = CallFinishSpawnNode->FindPinChecked(TransformParamName);
   552		UEdGraphPin* CallFinishScaleMethod = CallFinishSpawnNode->FindPinChecked(FK2Node_SpawnActorFromClassHelper::TransformScaleMethodPinName);
   553		UEdGraphPin* CallFinishResult = CallFinishSpawnNode->GetReturnValuePin();
   554	
   555		// Move 'then' connection from spawn node to 'finish spawn'
   556		CompilerContext.MovePinLinksToIntermediate(*SpawnNodeThen, *CallFinishThen);
   557	
   558		// Copy transform connection
   559		CompilerContext.CopyPinLinksToIntermediate(*CallBeginTransform, *CallFinishTransform);
   560	 
   561		// Copy 'ScaleMethod' connection
   562		CompilerContext.CopyPinLinksToIntermediate(*CallBeginScaleMethod, *CallFinishScaleMethod);
   563	
   564		// Connect output actor from 'begin' to 'finish'
   565		CallBeginResult->MakeLinkTo(CallFinishActor);
   566	
   567		// Move result connection from spawn node to 'finish spawn'
   568		CallFinishResult->PinType = SpawnNodeResult->PinType; // Copy type so it uses the right actor subclass
   569		CompilerContext.MovePinLinksToIntermediate(*SpawnNodeResult, *CallFinishResult);
   570	
   571		//////////////////////////////////////////////////////////////////////////
   572		// create 'set var' nodes
   573	
   574		// Get 'result' pin from 'begin spawn', this is the actual actor we want to set properties on
   575		UEdGraphPin* LastThen = FKismetCompilerUtilities::GenerateAssignmentNodes(CompilerContext, SourceGraph, CallBeginSpawnNode, SpawnNode, CallBeginResult, ClassToSpawn, CallBeginActorClassPin);
   576	
   577		// Make exec connection between 'then' on last node and 'finish'
   578		LastThen->MakeLinkTo(CallFinishExec);
   579	
   580		// Break any links to the expanded node
   581		SpawnNode->BreakAllNodeLinks();
   582	}

   157	
   158		// Break any links to the expanded node
   159		CreateWidgetNode->BreakAllNodeLinks();

    31	void UK2Node_AsyncAction::GetMenuActions(FBlueprintActionDatabaseRegistrar& ActionRegistrar) const
    32	{
    33		struct GetMenuActions_Utils
    34		{
    35			static void SetNodeFunc(UEdGraphNode* NewNode, bool /*bIsTemplateNode*/, TWeakObjectPtr<UFunction> FunctionPtr)
    36			{
    37				UK2Node_AsyncAction* AsyncTaskNode = CastChecked<UK2Node_AsyncAction>(NewNode);
    38				if (FunctionPtr.IsValid())
    39				{
    40					UFunction* Func = FunctionPtr.Get();
    41					FObjectProperty* ReturnProp = CastFieldChecked<FObjectProperty>(Func->GetReturnProperty());
    42							
    43					AsyncTaskNode->ProxyFactoryFunctionName = Func->GetFName();
    44					AsyncTaskNode->ProxyFactoryClass        = Func->GetOuterUClass();
    45					AsyncTaskNode->ProxyClass               = ReturnProp->PropertyClass;
    46				}
    47			}
    48		};
    49	
    50		UClass* NodeClass = GetClass();
    51		ActionRegistrar.RegisterClassFactoryActions<UBlueprintAsyncActionBase>(FBlueprintActionDatabaseRegistrar::FMakeFuncSpawnerDelegate::CreateLambda([NodeClass](const UFunction* FactoryFunc)->UBlueprintNodeSpawner*
    52		{
    53			UClass* FactoryClass = FactoryFunc ? FactoryFunc->GetOwnerClass() : nullptr;
    54			if (FactoryClass && FactoryClass->HasMetaData(TEXT("HasDedicatedAsyncNode")))
    55			{
    56				// Wants to use a more specific blueprint node to handle the async action
    57				return nullptr;
    58			}
    59	
    60			UBlueprintNodeSpawner* NodeSpawner = UBlueprintFunctionNodeSpawner::Create(FactoryFunc);
    61			check(NodeSpawner != nullptr);
    62			NodeSpawner->NodeClass = NodeClass;
    63	
    64			TWeakObjectPtr<UFunction> FunctionPtr = MakeWeakObjectPtr(const_cast<UFunction*>(FactoryFunc));
    65			NodeSpawner->CustomizeNodeDelegate = UBlueprintNodeSpawner::FCustomizeNodeDelegate::CreateStatic(GetMenuActions_Utils::SetNodeFunc, FunctionPtr);
    66	
    67			return NodeSpawner;
    68		}) );
    69	}
