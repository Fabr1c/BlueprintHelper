   653	//------------------------------------------------------------------------------
   654	static void BlueprintActionDatabaseImpl::AddClassFunctionActions(UClass const* const Class, FActionList& ActionListOut)
   655	{
   656		using namespace FBlueprintNodeSpawnerFactory; // for MakeMessageNodeSpawner()
   657		check(Class != nullptr);
   658	
   659		// loop over all the functions in the specified class; exclude-super because 
   660		// we can always get the super functions by looking up that class separately 
   661		for (TFieldIterator<UFunction> FunctionIt(Class, EFieldIteratorFlags::ExcludeSuper); FunctionIt; ++FunctionIt)
   662		{
   663			UFunction* Function = *FunctionIt;
   664	
   665			bool const bIsInheritedFunction = BlueprintActionDatabaseImpl::IsInheritedBlueprintFunction(Function);
   666			if (bIsInheritedFunction)
   667			{
   668				// inherited functions will be captured when the parent class is ran
   669				// through this function (no need to duplicate)
   670				continue;
   671			}
   672	
   673			// Apply general filtering for functions
   674			if(!FBlueprintActionDatabase::IsFunctionAllowed(Function, FBlueprintActionDatabase::EPermissionsContext::Node))
   675			{
   676				continue;
   677			}
   678	
   679			bool const bIsBpInterfaceFunc = BlueprintActionDatabaseImpl::IsBlueprintInterfaceFunction(Function);
   680			if (UEdGraphSchema_K2::FunctionCanBePlacedAsEvent(Function) && !bIsBpInterfaceFunc)
   681			{
   682				if (UBlueprintEventNodeSpawner* NodeSpawner = UBlueprintEventNodeSpawner::Create(Function))
   683				{
   684					ActionListOut.Add(NodeSpawner);
   685				}
   686			}
   687	
   688			// If this is a promotable function, and it has already been registered
   689			// than do NOT add it to the asset action database. We should
   690			// probably have some better logic for this, like adding our own node spawner
   691			const bool bIsRegisteredPromotionFunc =
   692				TypePromoDebug::IsTypePromoEnabled() &&
   693				FTypePromotion::IsFunctionPromotionReady(Function) &&
   694				FTypePromotion::IsOperatorSpawnerRegistered(Function);
   695	
   696			if (UEdGraphSchema_K2::CanUserKismetCallFunction(Function))
   697			{
   698				// @TODO: if this is a Blueprint, and this function is from a 
   699				//        Blueprint "implemented interface", then we don't need to 
   700				//        include it (the function is accounted for in from the 
   701				//        interface class).
   702				UBlueprintFunctionNodeSpawner* FuncSpawner = UBlueprintFunctionNodeSpawner::Create(Function);
   703				
   704				// Only add this action to the list of the operator function is not already registered. Otherwise we will 
   705				// get a bunch of duplicate operator actions
   706				if (!bIsRegisteredPromotionFunc)
   707				{
   708					ActionListOut.Add(FuncSpawner);
   709				}
   710	
   711				if (FKismetEditorUtilities::IsClassABlueprintInterface(Class))
   712				{
   713					// Use the default function name
   714					ActionListOut.Add(MakeMessageNodeSpawner(Function));
   715				}
   716			}
   717		}
   718	}
   719	
   720	//------------------------------------------------------------------------------
   800				UClass* NonConstClass = const_cast<UClass*>(Class);
   801				UBlueprintVariableNodeSpawner* GetterSpawner = UBlueprintVariableNodeSpawner::CreateFromMemberOrParam(UK2Node_VariableGet::StaticClass(), Property, nullptr, NonConstClass);
   802				ActionListOut.Add(GetterSpawner);
   803	//			UBlueprintVariableNodeSpawner* SetterSpawner = UBlueprintVariableNodeSpawner::CreateFromMemberOrParam(UK2Node_VariableSet::StaticClass(), Property);
   804	//			ActionListOut.Add(SetterSpawner);
   805			}
   806		}

   836	//------------------------------------------------------------------------------
   837	static void BlueprintActionDatabaseImpl::AddBlueprintGraphActions(UBlueprint const* const Blueprint, FActionList& ActionListOut)
   838	{
   839		using namespace FBlueprintNodeSpawnerFactory; // for MakeMacroNodeSpawner()
   840	
   841		for (UEdGraph* MacroGraph : Blueprint->MacroGraphs)
   842		{
   843			ActionListOut.Add(MakeMacroNodeSpawner(MacroGraph));
   844		}
   845	
   846		auto CreateEntriesForGraphLambda = [Blueprint, &ActionListOut](UEdGraph* FunctionGraph)
   847		{
   848			if (!FunctionGraph)
   849			{
   850				return;
   851			}
   852	
   853			TArray<UK2Node_FunctionEntry*> GraphEntryNodes;
   854			FunctionGraph->GetNodesOfClass<UK2Node_FunctionEntry>(GraphEntryNodes);
   855	
   856			for (UK2Node_FunctionEntry* FunctionEntry : GraphEntryNodes)
   857			{
   858				UFunction* SkeletonFunction = FindUField<UFunction>(Blueprint->SkeletonGeneratedClass, FunctionGraph->GetFName());
   859	
   860				// Create entries for function parameters
   861				if (SkeletonFunction != nullptr)
   862				{
   863					for (TFieldIterator<FProperty> ParamIt(SkeletonFunction); ParamIt && (ParamIt->PropertyFlags & CPF_Parm); ++ParamIt)
   864					{
   865						FProperty* Param = *ParamIt;
   866						const bool bIsFunctionInput = !Param->HasAnyPropertyFlags(CPF_ReturnParm) && (!Param->HasAnyPropertyFlags(CPF_OutParm) || Param->HasAnyPropertyFlags(CPF_ReferenceParm));
   867						if (bIsFunctionInput)
   868						{
   869							UBlueprintNodeSpawner* GetVarSpawner = UBlueprintVariableNodeSpawner::CreateFromMemberOrParam(UK2Node_VariableGet::StaticClass(), Param, FunctionGraph);
   870							ActionListOut.Add(GetVarSpawner);
   871						}
   872					}
   873				}
   874	
   875				// Create entries for local variables
   876				for (FBPVariableDescription const& LocalVar : FunctionEntry->LocalVariables)
   877				{
   878					// Create a member reference so we can safely resolve the FProperty
   879					FMemberReference Reference;
   880					Reference.SetLocalMember(LocalVar.VarName, FunctionGraph->GetName(), LocalVar.VarGuid);
   881	
   882					UBlueprintNodeSpawner* GetVarSpawner = UBlueprintVariableNodeSpawner::CreateFromLocal(UK2Node_VariableGet::StaticClass(), FunctionGraph, LocalVar, Reference.ResolveMember<FProperty>(Blueprint->SkeletonGeneratedClass));
   883					ActionListOut.Add(GetVarSpawner);
   884					UBlueprintNodeSpawner* SetVarSpawner = UBlueprintVariableNodeSpawner::CreateFromLocal(UK2Node_VariableSet::StaticClass(), FunctionGraph, LocalVar, Reference.ResolveMember<FProperty>(Blueprint->SkeletonGeneratedClass));
   885					ActionListOut.Add(SetVarSpawner);
   886				}
   887			}
   888		};
   889	
   890		auto CreateEntriesLambda = [&](const TArray<UEdGraph*>& Graphs)
   891		{
   892			for (UEdGraph* const Graph : Graphs)
   893			{
   894				CreateEntriesForGraphLambda(Graph);
   895			}
   896		};
   897	
   898		// local variables and parameters for functions
   899		CreateEntriesLambda(Blueprint->FunctionGraphs);
   900	
   901		// local variables and parameters for interfaces
   902		for (const FBPInterfaceDescription& Interface : Blueprint->ImplementedInterfaces)
   903		{
   904			CreateEntriesLambda(Interface.Graphs);
   905		}
   906	}
