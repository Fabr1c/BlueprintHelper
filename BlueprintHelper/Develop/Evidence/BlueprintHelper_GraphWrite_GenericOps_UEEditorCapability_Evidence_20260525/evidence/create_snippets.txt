   132			const bool bIsDelegate = Property->IsA(FMulticastDelegateProperty::StaticClass());
   133			const bool bIsExposedToSpawn = UEdGraphSchema_K2::IsPropertyExposedOnSpawn(Property);
   134			const bool bIsSettableExternally = !Property->HasAnyPropertyFlags(CPF_DisableEditOnInstance);
   135	
   136			if(	bIsExposedToSpawn &&

   140				!bIsDelegate &&

   144				if (UEdGraphPin* Pin = CreatePin(EGPD_Input, NAME_None, Property->GetFName()))

   150					}

    33	struct FK2Node_GenericCreateObject_Utils
    34	{
    35		static bool CanSpawnObjectOfClass(TSubclassOf<UObject> ObjectClass, bool bAllowAbstract)
    36		{
    37			// Initially include types that meet the basic requirements.
    38			// Note: CLASS_Deprecated is an inherited class flag, so any subclass of an explicitly-deprecated class also cannot be spawned.
    39			bool bCanSpawnObject = (nullptr != *ObjectClass)
    40				&& (bAllowAbstract || !ObjectClass->HasAnyClassFlags(CLASS_Abstract))
    41				&& !ObjectClass->HasAnyClassFlags(CLASS_Deprecated | CLASS_NewerVersionExists);
    42	
    43			// UObject is a special case where if we are allowing abstract we are going to allow it through even though it doesn't have BlueprintType on it
    44			if (bCanSpawnObject && (!bAllowAbstract || (*ObjectClass != UObject::StaticClass())))
    45			{
    46				static const FName BlueprintTypeName(TEXT("BlueprintType"));
    47				static const FName NotBlueprintTypeName(TEXT("NotBlueprintType"));
    48				static const FName DontUseGenericSpawnObjectName(TEXT("DontUseGenericSpawnObject"));
    49	
    50				auto IsClassAllowedLambda = [](const UClass* InClass)
    51				{
    52					return InClass != AActor::StaticClass()
    53						&& InClass != UActorComponent::StaticClass();
    54				};
    55	
    56				// Exclude all types in the initial set by default.
    57				bCanSpawnObject = false;
    58				const UClass* CurrentClass = ObjectClass;
    59	
    60				// Climb up the class hierarchy and look for "BlueprintType." If "NotBlueprintType" is seen first, or if the class is not allowed, then stop searching.
    61				while (!bCanSpawnObject && CurrentClass != nullptr && !CurrentClass->GetBoolMetaData(NotBlueprintTypeName) && IsClassAllowedLambda(CurrentClass))
    62				{
    63					// Include any type that either includes or inherits 'BlueprintType'
    64					bCanSpawnObject = CurrentClass->GetBoolMetaData(BlueprintTypeName);
    65	
    66					// Stop searching if we encounter 'BlueprintType' with 'DontUseGenericSpawnObject'
    67					if (bCanSpawnObject && CurrentClass->GetBoolMetaData(DontUseGenericSpawnObjectName))
    68					{
    69						bCanSpawnObject = false;
    70						break;
    71					}
    72	
    73					CurrentClass = CurrentClass->GetSuperClass();
    74				}
    75	
    76				// If we validated the given class, continue walking up the hierarchy to make sure we exclude it if it's an Actor or ActorComponent derivative.
    77				while (bCanSpawnObject && CurrentClass != nullptr)
    78				{
    79					bCanSpawnObject &= IsClassAllowedLambda(CurrentClass);
    80	
    81					CurrentClass = CurrentClass->GetSuperClass();
    82				}
    83			}
    84	
    85			return bCanSpawnObject;
    86		}
