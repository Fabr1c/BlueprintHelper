    41		{
    42			// Make sure that the input pin is connected and valid for this block
    43			FEdGraphPinType ExpectedPinType;
    44			ExpectedPinType.PinCategory = UEdGraphSchema_K2::PC_Exec;
    45	
    46			UEdGraphPin* ExecTriggeringPin = Context.FindRequiredPinByName(Node, UEdGraphSchema_K2::PN_Execute, EGPD_Input);
    47			if ((ExecTriggeringPin == nullptr) || !Context.ValidatePinType(ExecTriggeringPin, ExpectedPinType))
    48			{
    49				CompilerContext.MessageLog.Error(*LOCTEXT("NoValidExecutionPinForExecSeq_Error", "@@ must have a valid execution pin @@").ToString(), Node, ExecTriggeringPin);
    50				return;
    51			}
    52			else if (ExecTriggeringPin->LinkedTo.Num() == 0)
    53			{
    54				CompilerContext.MessageLog.Warning(*LOCTEXT("NodeNeverExecuted_Warning", "@@ will never be executed").ToString(), Node);
    55				return;
    56			}
    57	
    58			// Find the valid, connected output pins, and add them to the processing list
    59			TArray<UEdGraphPin*> OutputPins;
    60			for (UEdGraphPin* CurrentPin : Node->Pins)
    61			{
    62				if ((CurrentPin->Direction == EGPD_Output) && (CurrentPin->LinkedTo.Num() > 0) && (CurrentPin->PinName.ToString().StartsWith(UEdGraphSchema_K2::PN_Then.ToString())))
    63				{
    64					OutputPins.Add(CurrentPin);
    65				}
    66			}
    67	
    68			//@TODO: Sort the pins by the number appended to the pin!
    69	
    70			// Process the pins, if there are any valid entries
    71			if (OutputPins.Num() > 0)
    72			{
    73				if (Context.IsDebuggingOrInstrumentationRequired() && (OutputPins.Num() > 1))
    74				{
    75					const FString NodeComment = Node->NodeComment.IsEmpty() ? Node->GetName() : Node->NodeComment;
    76	
    77					// Assuming sequence X goes to A, B, C, we want to emit:
    78					//   X: push X1
    79					//      goto A
    80					//  X1: debug site
    81					//      push X2
    82					//      goto B
    83					//  X2: debug site
    84					//      goto C
    85	
    86					// A push statement we need to patch up on the next pass (e.g., push X1 before we know where X1 is)
    87					FBlueprintCompiledStatement* LastPushStatement = NULL;
    88	
    89					for (int32 i = 0; i < OutputPins.Num(); ++i)
    90					{
    91						// Emit the debug site and patch up the previous jump if we're on subsequent steps
    92						const bool bNotFirstIndex = i > 0;
    93						if (bNotFirstIndex)
    94						{
    95							// Emit a debug site
    96							FBlueprintCompiledStatement& DebugSiteAndJumpTarget = Context.AppendStatementForNode(Node);
    97							DebugSiteAndJumpTarget.Type = Context.GetBreakpointType();
    98							DebugSiteAndJumpTarget.Comment = NodeComment;
    99							DebugSiteAndJumpTarget.bIsJumpTarget = true;
   100	
   101							// Patch up the previous push jump target
   102							check(LastPushStatement);
   103							LastPushStatement->TargetLabel = &DebugSiteAndJumpTarget;
   104						}
   105	
   106						// Emit a push to get to the next step in the sequence, unless we're the last one or this is an instrumented build
   107						const bool bNotLastIndex = ((i + 1) < OutputPins.Num());
   108						if (bNotLastIndex)
   109						{
   110							FBlueprintCompiledStatement& PushExecutionState = Context.AppendStatementForNode(Node);
   111							PushExecutionState.Type = KCST_PushState;
   112							LastPushStatement = &PushExecutionState;
   113						}
   114	
   115						// Emit the goto to the actual state
   116						FBlueprintCompiledStatement& GotoSequenceLinkedState = Context.AppendStatementForNode(Node);
   117						GotoSequenceLinkedState.Type = KCST_UnconditionalGoto;
   118						Context.GotoFixupRequestMap.Add(&GotoSequenceLinkedState, OutputPins[i]);
   119					}
   120	
   121					check(LastPushStatement);
   122				}

   127					{
   128						FBlueprintCompiledStatement& PushExecutionState = Context.AppendStatementForNode(Node);

    39	void UK2Node_SwitchEnum::SetEnum(UEnum* InEnum)
    40	{
    41		Enum = InEnum;
    42	
    43		// regenerate enum name list
    44		EnumEntries.Empty();
    45		EnumFriendlyNames.Empty();
    46	
    47		if (Enum)
    48		{
    49			PreloadObject(Enum);
    50	
    51			// When on async loading thread, postload happens later on GT unless it's possible
    52			// to do it right now safely.
    53			if (IsInGameThread() || Enum->IsPostLoadThreadSafe())
    54			{
    55				Enum->ConditionalPostLoad();
    56			}
    57	
    58			for (int32 EnumIndex = 0; EnumIndex < Enum->NumEnums() - 1; ++EnumIndex)
    59			{
    60				bool const bShouldBeHidden = Enum->HasMetaData(TEXT("Hidden"), EnumIndex ) || Enum->HasMetaData(TEXT("Spacer"), EnumIndex );
    61				if (!bShouldBeHidden)
    62				{
    63					FString const EnumValueName = Enum->GetNameStringByIndex(EnumIndex);
    64					EnumEntries.Add( FName(*EnumValueName) );
    65	
    66					FText EnumFriendlyName = Enum->GetDisplayNameTextByIndex(EnumIndex);
    67					EnumFriendlyNames.Add( EnumFriendlyName );
    68				}
    69			}
    70		}
    71	}

   171	void UK2Node_SwitchEnum::CreateCasePins()
   172	{
   173		if (Enum)
   174		{
   175			SetEnum(Enum);
   176		}
   177	
   178		const bool bShouldUseAdvancedView = (EnumEntries.Num() > 5);
   179		if(bShouldUseAdvancedView && (ENodeAdvancedPins::NoPins == AdvancedPinDisplay))
   180		{
   181			AdvancedPinDisplay = ENodeAdvancedPins::Hidden;
   182		}
   183	
   184		for (int32 Index = 0; Index < EnumEntries.Num(); ++Index)
   185		{
   186			UEdGraphPin* NewPin = CreatePin(EGPD_Output, UEdGraphSchema_K2::PC_Exec, EnumEntries[Index]);
   187			if (EnumFriendlyNames.IsValidIndex(Index))
   188			{

   190			}
   191			
   192			if(bShouldUseAdvancedView && (Index > 2))
   193			{
   194				NewPin->bAdvancedView = true;
   195			}
   196		}

   199	UK2Node::ERedirectType UK2Node_SwitchEnum::DoPinsMatchForReconstruction(const UEdGraphPin* NewPin, int32 NewPinIndex, const UEdGraphPin* OldPin, int32 OldPinIndex) const
   200	{
   201		UK2Node::ERedirectType ReturnValue = Super::DoPinsMatchForReconstruction(NewPin, NewPinIndex, OldPin, OldPinIndex);
   202		if (ReturnValue == UK2Node::ERedirectType_None && Enum && OldPinIndex > 2 && NewPinIndex > 2)
   203		{
   204			const int64 OldValue = Enum->GetValueByName(OldPin->PinName);
   205			const int64 NewValue = Enum->GetValueByName(NewPin->PinName);
   206			// This handles redirects properly
   207			if (OldValue == NewValue && OldValue != INDEX_NONE)
   208			{
   209				ReturnValue = UK2Node::ERedirectType_Name;
   210			}
   211		}
   212		return ReturnValue;
