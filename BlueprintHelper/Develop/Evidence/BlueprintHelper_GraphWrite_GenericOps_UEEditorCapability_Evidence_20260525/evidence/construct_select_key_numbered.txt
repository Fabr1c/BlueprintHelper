   104	static bool CanBeExposed(const FProperty* Property, UBlueprint* BP)
   105	{
   106		if (Property)
   107		{
   108			const UEdGraphSchema_K2* Schema = GetDefault<UEdGraphSchema_K2>();
   109			check(Schema);
   110	
   111			// Treat all inline edit condition properties as override flags; that is, don't allow
   112			// these to be exposed as part of the optional input pin set. Their value will be set
   113			// implicitly at runtime based on whether or not any bound members are exposed, rather
   114			// than explicitly via an exposed input pin. This emulates how the Property Editor
   115			// handles setting these values at edit time (they appear as an inline checkbox that
   116			// the user ticks on to set the flag and enable/override a bound property's value).
   117			static const FName MD_InlineEditConditionToggle(TEXT("InlineEditConditionToggle"));
   118			if (Property->HasMetaData(MD_InlineEditConditionToggle))
   119			{
   120				return false;
   121			}
   122	
   123			const bool bIsEditorBP = IsEditorOnlyObject(BP);
   124			const bool bIsEditAnywhereProperty = Property->HasAllPropertyFlags(CPF_Edit) &&
   125				!Property->HasAnyPropertyFlags(CPF_EditConst);
   126	
   127			if (!Property->HasAllPropertyFlags(CPF_BlueprintReadOnly) || 
   128				(bIsEditorBP && bIsEditAnywhereProperty) )
   129			{
   130				if (Property->HasAllPropertyFlags(CPF_BlueprintVisible) && !(Property->ArrayDim > 1))
   131				{
   132					FEdGraphPinType DumbGraphPinType;
   133					if (Schema->ConvertPropertyToPinType(Property, /*out*/ DumbGraphPinType))
   134					{
   135						return true;
   136					}
   137				}
   138			}
   139		}
   140		return false;

   265	{
   266		if (FromPin)
   267		{
   268			// Attempt to autowire to the index pin as users generally drag off of something intending to use

   270			const UEdGraphSchema_K2* K2Schema = CastChecked<UEdGraphSchema_K2>(GetSchema());
   335	void UK2Node_Select::ReallocatePinsDuringReconstruction(TArray<UEdGraphPin*>& OldPins)
   336	{
   337		Super::ReallocatePinsDuringReconstruction(OldPins);
   338	
   339		const UEdGraphSchema_K2* Schema = GetDefault<UEdGraphSchema_K2>();
   340	
   341		// See if this node was saved in the old version with a boolean as the condition
   342		UEdGraphPin* OldConditionPin = nullptr;
   343		UEdGraphPin* OldIndexPin = nullptr;
   344		UEdGraphPin* OldReturnPin = nullptr;
   345		for (UEdGraphPin* OldPin : OldPins)
   346		{
   347			if (OldPin->PinName == NAME_bPickOption0)
   348			{
   349				OldConditionPin = OldPin;
   350			}
   351			else if (OldPin->PinName == NAME_Index)
   352			{
   353				OldIndexPin = OldPin;
   354			}
   355			else if (OldPin->PinName == UEdGraphSchema_K2::PN_ReturnValue)
   356			{
   357				OldReturnPin = OldPin;
   358			}
   359		}
   360	
   361		UEdGraphPin* ReturnPin = GetReturnValuePin();
   362		check(ReturnPin);
   363	
   364		if (OldReturnPin && (ReturnPin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard))
   365		{
   366			// Always copy type from node prior, if pins have changed those will error at compilation time
   367			ReturnPin->PinType = OldReturnPin->PinType;
   368		}
   369	
   370		UEdGraphPin* IndexPin = GetIndexPin();
   371		check(IndexPin);
   372	
   373		// If we are fixing up an old bool node (swap the options and copy the condition links)
   374		if (OldConditionPin)
   375		{
   376			// Set the index pin type
   377			IndexPinType.PinCategory = UEdGraphSchema_K2::PC_Boolean;
   378			IndexPinType.PinSubCategory = NAME_None;
   379			IndexPinType.PinSubCategoryObject = nullptr;
   380	
   381			// Set the pin type and Copy the pin
   382			IndexPin->PinType = IndexPinType;
   383			Schema->CopyPinLinks(*OldConditionPin, *IndexPin);
   384			// If we copy links, we need to send a notification
   385			if (IndexPin->LinkedTo.Num() > 0)
   386			{
   387				PinConnectionListChanged(IndexPin);
   388			}
   389	
   390			UEdGraphPin* OptionPin0 = FindPin(NAME_Option_0);
   391			UEdGraphPin* OptionPin1 = FindPin(NAME_Option_1);
   392	
   393			for (UEdGraphPin* OldPin : OldPins)
   394			{
   395				if (OldPin->PinName == OptionPin0->PinName)
   396				{
   397					Schema->MovePinLinks(*OldPin, *OptionPin1);
   398				}
   399				else if (OldPin->PinName == OptionPin1->PinName)
   400				{
   401					Schema->MovePinLinks(*OldPin, *OptionPin0);
   402				}
   403			}
   404		}
   405	
   406		// If the index pin has links or a default value but is a wildcard, this is an old int pin so convert it
   407		if (OldIndexPin &&
   408			IndexPinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard &&
   409			(OldIndexPin->LinkedTo.Num() > 0 || !OldIndexPin->DefaultValue.IsEmpty()))
   410		{
   411			IndexPinType.PinCategory = UEdGraphSchema_K2::PC_Int;
   412			IndexPinType.PinSubCategory = NAME_None;
   413			IndexPinType.PinSubCategoryObject = nullptr;
   414			IndexPin->PinType = IndexPinType;
   415		}
   416	
   417		// Set up default values for index and option pins now that the information is available
   418		Schema->SetPinAutogeneratedDefaultValueBasedOnType(IndexPin);
   419	
   420		const bool bFillTypeFromReturn = ReturnPin->PinType.PinCategory != UEdGraphSchema_K2::PC_Wildcard;
   421		TArray<UEdGraphPin*> OptionPins;
   422		GetOptionPins(OptionPins);
   423		for (UEdGraphPin* Pin : OptionPins)
   424		{
   425			const bool bTypeShouldBeFilled = Pin && (Pin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard);
   426			if (bTypeShouldBeFilled && bFillTypeFromReturn)
   427			{
   428				Pin->PinType = ReturnPin->PinType;			
   429			}
   430			Schema->SetPinAutogeneratedDefaultValueBasedOnType(Pin);
   431		}
   432	}

   434	void UK2Node_Select::PostReconstructNode()
   435	{
   436		// After ReconstructNode we must be sure that no additional reconstruction is required
   437		bReconstructNode = false;
   438		bReconstructForPinTypeChange = false;
   439	
   440		UEdGraphPin* ReturnPin = GetReturnValuePin();
   441		const bool bFillTypeFromConnected = ReturnPin && (ReturnPin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard);
   442	
   443		if (bFillTypeFromConnected)
   444		{
   445			FEdGraphPinType PinType = ReturnPin->PinType;
   446	
   447			if (ReturnPin->LinkedTo.Num() > 0)
   448			{
   449				PinType = ReturnPin->LinkedTo[0]->PinType;
   450			}
   451			else
   452			{
   453				TArray<UEdGraphPin*> OptionPins;
   454				GetOptionPins(OptionPins);
   455				for (UEdGraphPin* Pin : OptionPins)
   456				{
   457					if (Pin && Pin->LinkedTo.Num() > 0)
   458					{
   459						PinType = Pin->LinkedTo[0]->PinType;
   460						break;
   461					}
   462				}
   463			}
   464	
   465			ReturnPin->PinType = PinType;
   466			OnPinTypeChanged(ReturnPin);
   467		}
   468	
   469		Super::PostReconstructNode();
   470	}

   472	/** Determine if any pins are connected, if so make all the other pins the same type, if not, make sure pins are switched back to wildcards */
   473	void UK2Node_Select::NotifyPinConnectionListChanged(UEdGraphPin* Pin)
   474	{
   475		Super::NotifyPinConnectionListChanged(Pin);
   476	
   477		// If this is the Enum pin we need to set the enum and reconstruct the node
   478		if (Pin == GetIndexPin())
   479		{
   480			// If the index pin was just linked to another pin
   481			if (Pin->LinkedTo.Num() > 0 && Pin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard)
   482			{
   483				UEdGraphPin* LinkPin = Pin->LinkedTo[0];
   484	
   485				if (Pin->PinType != LinkPin->PinType)
   486				{
   487					Pin->PinType = LinkPin->PinType;
   488	
   489					OnPinTypeChanged(Pin);
   490				}
   491			}
   492		}
   493		else
   494		{
   495			// Grab references to all option pins and the return pin
   496			TArray<UEdGraphPin*> OptionPins;
   497			GetOptionPins(OptionPins);
   498			UEdGraphPin* ReturnPin = FindPin(UEdGraphSchema_K2::PN_ReturnValue);
   499	
   500			// See if this pin is one of the wildcard pins
   501			const bool bIsWildcardPin = ((Pin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard) && ((Pin == ReturnPin) || (OptionPins.Find(Pin) != INDEX_NONE)));
   502	
   503			TFunction<bool(UEdGraphPin*)> PinInUse = [&PinInUse](UEdGraphPin* PinToConsider)
   504			{
   505				bool bPinInUse = ((PinToConsider->LinkedTo.Num() > 0) || (PinToConsider->ParentPin != nullptr) || !PinToConsider->DoesDefaultValueMatchAutogenerated());
   506				if (!bPinInUse)
   507				{
   508					for (UEdGraphPin* SubPin : PinToConsider->SubPins)
   509					{
   510						bPinInUse = PinInUse(SubPin);
   511						if (bPinInUse)
   512						{
   513							break;
   514						}
   515					}
   516				}
   517				return bPinInUse;
   518			};
   519	
   520			bool bPinsInUse = PinInUse(ReturnPin);
   521			
   522			if (!bPinsInUse)
   523			{
   524				for (UEdGraphPin* OptionPin : OptionPins)
   525				{
   526					bPinsInUse = PinInUse(OptionPin);
   527					if (bPinsInUse)
   528					{
   529						break;
   530					}
   531				}
   532			}
   533	
   534			bool bPinTypeChanged = false;
   535	
   536			if (bPinsInUse)
   537			{
   538				// If the pin was one of the wildcards we have to handle it specially
   539				if (bIsWildcardPin)
   540				{
   541					// If the pin is linked, make sure the other wildcard pins match
   542					if (Pin->LinkedTo.Num() > 0)
   543					{
   544						UEdGraphPin* LinkPin = Pin->LinkedTo[0];
   545	
   546						if (Pin->PinType != LinkPin->PinType)
   547						{
   548							Pin->PinType = LinkPin->PinType;
   549							bPinTypeChanged = true;
   550						}
   551					}
   552				}
   553			}
   554			else
   555			{
   556				bPinTypeChanged = true;
   557				Pin->PinType.PinCategory = UEdGraphSchema_K2::PC_Wildcard;
   558				Pin->PinType.PinSubCategory = NAME_None;
   559				Pin->PinType.PinSubCategoryObject = nullptr;
   560			}
   561	
   562			if (bPinTypeChanged)
   563			{
   564				OnPinTypeChanged(Pin);
   565			}
   566		}
   567	}

   808	void UK2Node_Select::OnPinTypeChanged(UEdGraphPin* Pin)
   809	{
   810		const UEdGraphSchema_K2* Schema = GetDefault<UEdGraphSchema_K2>();
   811	
   812		if (Pin == GetIndexPin())
   813		{
   814			if (IndexPinType != Pin->PinType)
   815			{
   816				IndexPinType = Pin->PinType;
   817	
   818				// Since it is an interactive action we want the pins to go away regardless of the new type
   819				for (UEdGraphPin* PinToDiscard : Pins)
   820				{
   821					PinToDiscard->SetSavePinIfOrphaned(false);
   822				}
   823	
   824				if (IndexPinType.PinSubCategoryObject.IsValid())
   825				{
   826					SetEnum(Cast<UEnum>(IndexPinType.PinSubCategoryObject.Get()));
   827				}
   828				else if (Enum)
   829				{
   830					SetEnum(nullptr);
   831				}
   832	
   833				// Remove all but two options if we switched to a bool index
   834				if (IndexPinType.PinCategory == UEdGraphSchema_K2::PC_Boolean)
   835				{
   836					NumOptionPins = 2;
   837				}
   838	
   839				if (!Schema->IsPinDefaultValid(Pin, Pin->DefaultValue, Pin->DefaultObject, Pin->DefaultTextValue).IsEmpty())
   840				{
   841					Schema->ResetPinToAutogeneratedDefaultValue(Pin);
   842				}
   843	
   844				bReconstructNode = true;
   845			}
   846		}
   847		else
   848		{
   849			// Set the return value
   850			UEdGraphPin* ReturnPin = GetReturnValuePin();
   851		
   852			// Recombine the sub pins back into the ReturnPin
   853			if (ReturnPin->SubPins.Num() > 0)
   854			{
   855				Schema->RecombinePin(ReturnPin->SubPins[0]);
   856			}
   857			ReturnPin->PinType = Pin->PinType;
   858	
   859			// Recombine all option pins back into their root
   860			TArray<UEdGraphPin*> OptionPins;
   861	 		GetOptionPins(OptionPins);
   862			for (UEdGraphPin* OptionPin : OptionPins)
   863			{
   864				// Recombine the sub pins back into the OptionPin
   865				if (OptionPin->ParentPin == nullptr && OptionPin->SubPins.Num() > 0)
   866				{
   867					Schema->RecombinePin(OptionPin->SubPins[0]);
   868				}
   869			}
   870	
   871			// Get the options again and set them
   872			GetOptionPins(OptionPins);
   873			for (UEdGraphPin* OptionPin : OptionPins)
   874			{
   875				if (OptionPin->PinType != Pin->PinType ||
   876					OptionPin == Pin)
   877				{
   878					OptionPin->PinType = Pin->PinType;
   879				}
   880	
   881				if (!Schema->IsPinDefaultValid(OptionPin, OptionPin->DefaultValue, OptionPin->DefaultObject, OptionPin->DefaultTextValue).IsEmpty())
   882				{
   883					Schema->ResetPinToAutogeneratedDefaultValue(OptionPin);
   884				}
   885			}
   886	
   887			bReconstructNode = true;
   888		}
   889	}

   930	void UK2Node_Select::GetMenuActions(FBlueprintActionDatabaseRegistrar& ActionRegistrar) const
   931	{
   932		// actions get registered under specific object-keys; the idea is that 
   933		// actions might have to be updated (or deleted) if their object-key is  
   934		// mutated (or removed)... here we use the node's class (so if the node 
   935		// type disappears, then the action should go with it)
   936		UClass* ActionKey = GetClass();
   937		// to keep from needlessly instantiating a UBlueprintNodeSpawner, first   
   938		// check to make sure that the registrar is looking for actions of this type
   939		// (could be regenerating actions for a specific asset, and therefore the 
   940		// registrar would only accept actions corresponding to that asset)
   941		if (ActionRegistrar.IsOpenForRegistration(ActionKey))
   942		{
   943			UBlueprintNodeSpawner* NodeSpawner = UBlueprintNodeSpawner::Create(GetClass());
   944			check(NodeSpawner != nullptr);
   945	
   946			ActionRegistrar.AddBlueprintAction(ActionKey, NodeSpawner);
   947		}
