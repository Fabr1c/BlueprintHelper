   104	static bool CanBeExposed(const FProperty* Property, UBlueprint* BP)
   105	{
   106		if (Property)
   107		{
   108			const UEdGraphSchema_K2* Schema = GetDefault<UEdGraphSchema_K2>();
   109			check(Schema);
   110	
   111			// Treat all inline edit condition properties as override flags; that is, don't allow
   112			// these to be exposed as part of the optional input pin set. Their value will be set
   113			// implicitly at runtime based on whether or not any bound members are exposed, rather
   114			// than explicitly via an exposed input pin. This emulates how the Property Editor
   115			// handles setting these values at edit time (they appear as an inline checkbox that
   116			// the user ticks on to set the flag and enable/override a bound property's value).
   117			static const FName MD_InlineEditConditionToggle(TEXT("InlineEditConditionToggle"));
   118			if (Property->HasMetaData(MD_InlineEditConditionToggle))
   119			{
   120				return false;
   121			}
   122	
   123			const bool bIsEditorBP = IsEditorOnlyObject(BP);
   124			const bool bIsEditAnywhereProperty = Property->HasAllPropertyFlags(CPF_Edit) &&
   125				!Property->HasAnyPropertyFlags(CPF_EditConst);
   126	
   127			if (!Property->HasAllPropertyFlags(CPF_BlueprintReadOnly) || 
   128				(bIsEditorBP && bIsEditAnywhereProperty) )
   129			{
   130				if (Property->HasAllPropertyFlags(CPF_BlueprintVisible) && !(Property->ArrayDim > 1))
   131				{
   132					FEdGraphPinType DumbGraphPinType;
   133					if (Schema->ConvertPropertyToPinType(Property, /*out*/ DumbGraphPinType))
   134					{
   135						return true;
   136					}
   137				}
   138			}
   139		}
   140		return false;

  7206		UEdGraph* Graph = GraphNode->GetGraph();
  7207		UScriptStruct* StructType = Cast<UScriptStruct>(Pin->PinType.PinSubCategoryObject.Get());
  7208		if (!StructType)
  7209		{
  7210			if (Params.CompilerContext)
  7211			{
  7212				Params.CompilerContext->MessageLog.Error(TEXT("No structure in SubCategoryObject in pin @@"), Pin);
  7213			}
  7214			StructType = GetFallbackStruct();
  7215		}
  7216	
  7217		UK2Node* SplitPinNode = nullptr;
  7218	
  7219		if (Pin->Direction == EGPD_Input)
  7220		{
  7221			if (UK2Node_MakeStruct::CanBeMade(StructType))
  7222			{
  7223				UK2Node_MakeStruct* MakeStructNode;
  7224	
  7225				if (Params.bTransient || Params.CompilerContext)
  7226				{
  7227					MakeStructNode = (Params.bTransient ? NewObject<UK2Node_MakeStruct>(Graph) : Params.CompilerContext->SpawnIntermediateNode<UK2Node_MakeStruct>(GraphNode, Params.SourceGraph));
  7228					MakeStructNode->StructType = StructType;
  7229					MakeStructNode->bMadeAfterOverridePinRemoval = true;
  7230					MakeStructNode->AllocateDefaultPins();
  7231				}
  7232				else
  7233				{
  7234					FGraphNodeCreator<UK2Node_MakeStruct> MakeStructCreator(*Graph);
  7235					MakeStructNode = MakeStructCreator.CreateNode(false);
  7236					MakeStructNode->StructType = StructType;
  7237					MakeStructNode->bMadeAfterOverridePinRemoval = true;
  7238					MakeStructCreator.Finalize();
  7239				}
  7240	
  7241				if (Pin->DefaultValue.Len() > 0)
  7242				{
  7243					FStructOnScope StructOnScope(StructType);
  7244					StructType->ImportText(*Pin->DefaultValue, StructOnScope.GetStructMemory(), nullptr, PPF_None, GLog, StructType->GetName());
  7245	
  7246					for (TFieldIterator<FProperty> PropIt(StructType); PropIt; ++PropIt)
  7247					{
  7248						if (UEdGraphPin* MakeStructPin = MakeStructNode->FindPin(PropIt->GetFName(), EGPD_Input))
  7249						{
  7250							MakeStructPin->DefaultValue.Reset();
  7251							FBlueprintEditorUtils::PropertyValueToString(*PropIt, StructOnScope.GetStructMemory(), MakeStructPin->DefaultValue);
  7252							MakeStructPin->AutogeneratedDefaultValue = MakeStructPin->DefaultValue;
  7253						}
  7254					}
  7255				}
  7256	
  7257				SplitPinNode = MakeStructNode;
  7258			}
  7259			else
  7260			{
  7261				const FString& MetaData = StructType->GetMetaData(FBlueprintMetadata::MD_NativeMakeFunction);
  7262				const UFunction* Function = FindObject<UFunction>(nullptr, *MetaData, true);
  7263	
  7264				UK2Node_CallFunction* CallFunctionNode;
  7265	
  7266				if (Params.bTransient || Params.CompilerContext)
  7267				{
  7268					CallFunctionNode = (Params.bTransient ? NewObject<UK2Node_CallFunction>(Graph) : Params.CompilerContext->SpawnIntermediateNode<UK2Node_CallFunction>(GraphNode, Params.SourceGraph));
  7269					CallFunctionNode->SetFromFunction(Function);
  7270					CallFunctionNode->AllocateDefaultPins();
  7271				}
  7272				else
  7273				{
  7274					FGraphNodeCreator<UK2Node_CallFunction> MakeStructCreator(*Graph);
  7275					CallFunctionNode = MakeStructCreator.CreateNode(false);
  7276					CallFunctionNode->SetFromFunction(Function);
  7277					MakeStructCreator.Finalize();
  7278				}
  7279	
  7280				SplitPinNode = CallFunctionNode;
  7281			}
  7282		}
  7283		else
  7284		{
  7285			if (UK2Node_BreakStruct::CanBeBroken(StructType))
  7286			{
  7287				UK2Node_BreakStruct* BreakStructNode;
  7288	
  7289				if (Params.bTransient || Params.CompilerContext)
  7290				{
  7291					BreakStructNode = (Params.bTransient ? NewObject<UK2Node_BreakStruct>(Graph) : Params.CompilerContext->SpawnIntermediateNode<UK2Node_BreakStruct>(GraphNode, Params.SourceGraph));
  7292					BreakStructNode->StructType = StructType;
  7293					BreakStructNode->bMadeAfterOverridePinRemoval = true;
  7294					BreakStructNode->AllocateDefaultPins();
  7295				}
  7296				else
  7297				{
  7298					FGraphNodeCreator<UK2Node_BreakStruct> MakeStructCreator(*Graph);
  7299					BreakStructNode = MakeStructCreator.CreateNode(false);
  7300					BreakStructNode->StructType = StructType;
  7301					BreakStructNode->bMadeAfterOverridePinRemoval = true;
  7302					MakeStructCreator.Finalize();
  7303				}
  7304	
  7305				SplitPinNode = BreakStructNode;
  7306			}
  7307			else
  7308			{
  7309				const FString& MetaData = StructType->GetMetaData(FBlueprintMetadata::MD_NativeBreakFunction);
  7310				const UFunction* Function = FindObject<UFunction>(nullptr, *MetaData, true);
  7311	
  7312				UK2Node_CallFunction* CallFunctionNode;
  7313	
  7314				if (Params.bTransient || Params.CompilerContext)
  7315				{
  7316					CallFunctionNode = (Params.bTransient ? NewObject<UK2Node_CallFunction>(Graph) : Params.CompilerContext->SpawnIntermediateNode<UK2Node_CallFunction>(GraphNode, Params.SourceGraph));
  7317					CallFunctionNode->SetFromFunction(Function);
  7318					CallFunctionNode->AllocateDefaultPins();
  7319				}
  7320				else
  7321				{
  7322					FGraphNodeCreator<UK2Node_CallFunction> MakeStructCreator(*Graph);
  7323					CallFunctionNode = MakeStructCreator.CreateNode(false);
  7324					CallFunctionNode->SetFromFunction(Function);
  7325					MakeStructCreator.Finalize();
  7326				}
  7327	
  7328				SplitPinNode = CallFunctionNode;
  7329			}
  7330		}
  7331	
  7332		SplitPinNode->NodePosX = GraphNode->NodePosX - SplitPinNode->NodeWidth - 10;
  7333		SplitPinNode->NodePosY = GraphNode->NodePosY;
  7334	
  7335		return SplitPinNode;
  7336	}

  7339	{

  7346	
  7347		UEdGraphNode* GraphNode = Pin->GetOwningNode();
