   172	void UK2Node_MakeContainer::ClearPinTypeToWildcard()
   173	{
   174		if (CanResetToWildcard())
   175		{
   176			UEdGraphPin* OutputPin = GetOutputPin();
   177			OutputPin->PinType.PinCategory = UEdGraphSchema_K2::PC_Wildcard;
   178			OutputPin->PinType.PinSubCategory = NAME_None;
   179			OutputPin->PinType.PinSubCategoryObject = nullptr;
   180	
   220		bool bNotifyGraphChanged = false;
   221		if (Pin->LinkedTo.Num() > 0)
   222		{
   223			if (Pin->ParentPin == nullptr)
   224			{
   225				if (Pin == OutputPin)
   226				{
   227					if (NumKeyPinsWithLinks == 0 && (OutputPin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard && Pin->LinkedTo[0]->PinType.PinCategory != UEdGraphSchema_K2::PC_Wildcard))
   228					{
   229						FEdGraphTerminalType TerminalType = MoveTemp(OutputPin->PinType.PinValueType);
   230						OutputPin->PinType = Pin->LinkedTo[0]->PinType;
   231						OutputPin->PinType.PinValueType = MoveTemp(TerminalType);
   232						OutputPin->PinType.ContainerType = ContainerType;
   233						bNotifyGraphChanged = true;
   234					}
   235					if (ContainerType == EPinContainerType::Map && NumValuePinsWithLinks == 0 && (OutputPin->PinType.PinValueType.TerminalCategory == UEdGraphSchema_K2::PC_Wildcard && Pin->LinkedTo[0]->PinType.PinValueType.TerminalCategory != UEdGraphSchema_K2::PC_Wildcard))
   236					{
   237						OutputPin->PinType.PinValueType = Pin->LinkedTo[0]->PinType.PinValueType;
   238						bNotifyGraphChanged = true;
   239					}
   240				}
   241				else if (ValuePins.Contains(Pin))
   242				{
   243					// Just made a connection to a value pin, was it the first?
   244					if (NumValuePinsWithLinks == 1 && (OutputPin->PinType.PinValueType.TerminalCategory == UEdGraphSchema_K2::PC_Wildcard && Pin->LinkedTo[0]->PinType.PinCategory != UEdGraphSchema_K2::PC_Wildcard))
   245					{
   246						// Update the types on all the pins
   247						OutputPin->PinType.PinValueType = FEdGraphTerminalType::FromPinType(Pin->LinkedTo[0]->PinType);
   248						bNotifyGraphChanged = true;
   249					}
   250				}
   251				else 
   252				{
   253					// Just made a connection to a key pin, was it the first?
   254					if (NumKeyPinsWithLinks == 1 && (OutputPin->PinType.PinCategory == UEdGraphSchema_K2::PC_Wildcard && Pin->LinkedTo[0]->PinType.PinCategory != UEdGraphSchema_K2::PC_Wildcard))
   255					{
   256						FEdGraphTerminalType TerminalType = MoveTemp(OutputPin->PinType.PinValueType);
   257						OutputPin->PinType = Pin->LinkedTo[0]->PinType;
   258						OutputPin->PinType.PinValueType = MoveTemp(TerminalType);
   259						OutputPin->PinType.ContainerType = ContainerType;
   260						bNotifyGraphChanged = true;
   261					}
   262				}
   263			}
   264		}
   265		else if (OutputPin->LinkedTo.Num() == 0)
   266		{
   267			// Return to wildcard if theres nothing in any of the input pins
   268			TFunction<bool(TArray<UEdGraphPin*>&)> PinsInUse = [this, &PinsInUse](TArray<UEdGraphPin*>& PinsToConsider)
   269			{
   270				bool bPinInUse = false;
   271				for (UEdGraphPin* CurrentPin : PinsToConsider)
   272				{
   273					// Is there something in this pin?
   274					if (CurrentPin->SubPins.Num() > 0 || !CurrentPin->DoesDefaultValueMatchAutogenerated())
   275					{
   276						bPinInUse = true;
   277						break;
   278					}
   279				}
   280				return bPinInUse;
   281			};
   282	
   283			const bool bResetOutputPinPrimary = ((NumKeyPinsWithLinks == 0) && !PinsInUse(KeyPins));
   284			const bool bResetOutputPinSecondary = ((NumValuePinsWithLinks == 0) && !PinsInUse(ValuePins));
   285	
   286			if (bResetOutputPinPrimary)
   287			{
   288				OutputPin->PinType.PinCategory = UEdGraphSchema_K2::PC_Wildcard;
   289				OutputPin->PinType.PinSubCategory = NAME_None;
   290				OutputPin->PinType.PinSubCategoryObject = nullptr;
   291	
   292				bNotifyGraphChanged = true;
   293			}
   294			if (bResetOutputPinSecondary && ContainerType == EPinContainerType::Map)
   295			{
   296				OutputPin->PinType.PinValueType.TerminalCategory = UEdGraphSchema_K2::PC_Wildcard;
   297				OutputPin->PinType.PinValueType.TerminalSubCategory = NAME_None;
   298				OutputPin->PinType.PinValueType.TerminalSubCategoryObject = nullptr;
   299	
   300				bNotifyGraphChanged = true;
   301			}
   302		}
   303	
   304		if (bNotifyGraphChanged)
   305		{
   306			PropagatePinType();
   307			GetGraph()->NotifyNodeChanged(this);
   308		}
   309	}
